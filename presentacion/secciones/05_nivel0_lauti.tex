\section{Nivel 0 --- Protección}

\begin{frame}{Nivel 0: capa de máxima autoridad}
  \begin{itemize}
    \item Supervisa la planta de forma \textbf{permanente} (redundancia de sensores).
    \item Ante anomalía, \textbf{sobrescribe} a los niveles superiores y aplica los frenos.
    \item Recomputa \sig{vh\_lim} de forma \textbf{independiente} (defensa en profundidad).
  \end{itemize}
  \begin{alertblock}{Condiciones de emergencia}
    Sobrevelocidad \;$\bullet$\; fines de carrera de emergencia \;$\bullet$\; parada manual
    (botón) \;$\bullet$\; pérdida de \textit{heartbeat}.
  \end{alertblock}
\end{frame}

\begin{frame}{Autómata de seguridad: frenado selectivo}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{itemize}
        \item Estado nominal \texttt{SISTEMA\_OK} (ambos frenos abiertos).
        \item \textbf{Frenado selectivo}: sólo se detiene el \textbf{eje afectado}.
          \begin{itemize}
            \item Falla izaje $\Rightarrow$ frena izaje, carro libre.
            \item Falla carro $\Rightarrow$ frena carro, izaje libre.
            \item Emergencia manual $\Rightarrow$ ambos.
          \end{itemize}
        \item Rearme manual $\to$ \texttt{RECUPERACION}.
      \end{itemize}
    \end{column}
    \begin{column}{0.55\textwidth}
      \centering
      \includegraphics[width=\linewidth]{nivel_0/estados_seguridad.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Monitor de comunicaciones (watchdog)}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Persigue el \textbf{heartbeat} del supervisor.
        \item Si el contador supera el límite seguro $\Rightarrow$ \sig{FALLA\_WD}.
        \item Dispara la parada de emergencia: la grúa nunca opera con un comando
              \textbf{desactualizado}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\linewidth]{nivel_0/watchdog.png}
    \end{column}
  \end{columns}
\end{frame}
