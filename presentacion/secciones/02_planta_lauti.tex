\begin{frame}{Subsistema de izaje: visión general}
  \begin{block}{Función del sistema}
    Controla la elevación de la carga mediante:
    \begin{itemize}
      \item Motor eléctrico accionando un \textbf{tambor}
      \item Sistema de cableado con relación de transmisión \textbf{2:1} (aparejo móvil)
      \item Carga suspendida: \alert{2 grados de libertad} (traslación horizontal $x_l$ y vertical $y_l$)
    \end{itemize}
  \end{block}
  
  \vspace{0.3cm}
  \begin{alertblock}{Característica clave}
    Dinámica \textbf{híbrida}: parámetros cambian discretamente según el estado de los twistlocks (TLK)
  \end{alertblock}
\end{frame}

\begin{frame}{Dinámica de la carga: parámetros inerciales}
  \begin{block}{Masa total ($m_l$) — depende de TLK}
    \[
    m_l(\text{TLK}) = 
    \begin{cases} 
      M_s = 15000 \text{ kg} & \text{si TLK} = 1 \text{ (solo spreader)} \\[6pt]
      M_s + M_{cX} & \text{si TLK} = 0 \text{ (spreader + contenedor)}
    \end{cases}
    \]
    \vspace{-0.3cm}
    \begin{itemize}
      \item $M_{c\min} = 2000$ kg (contenedor vacío)
      \item $M_{c\max} = 50000$ kg (carga máxima)
    \end{itemize}
  \end{block}

  \begin{block}{Altura de la carga ($h_c$) — norma ISO}
    \[
    h_c(\text{TLK}) = 
    \begin{cases} 
      0 & \text{si TLK} = 1 \\
      H_c = 2.59 \text{ m} & \text{si TLK} = 0
    \end{cases}
    \]
  \end{block}
\end{frame}

\begin{frame}{Ecuaciones de movimiento de la carga}
  \begin{block}{Dinámicas acopladas — 2 DOF}
    \vspace{-0.15cm}
    \begin{equation*}
      m_l \, \frac{dv_{lx}}{dt} = -2 F_{hw} \sin\theta_l + F_{cx}
    \end{equation*}
    \vspace{-0.4cm}
    \begin{equation*}
      m_l \, \frac{dv_{ly}}{dt} = 2 F_{hw} \cos\theta_l - m_l g + F_{cy}
    \end{equation*}
    \begin{itemize}
      \item $\theta_l$: ángulo de balanceo del cable
      \item $F_{hw}$: fuerza de tensión del cable (elástica + amortiguadora)
      \item $F_{cx}$, $F_{cy}$: fuerzas de contacto con el perfil de obstáculos
    \end{itemize}
  \end{block}

  \begin{alertblock}{Acoplamiento}
    Los movimientos horizontal y vertical están acoplados a través de la geometría del cable y el ángulo $\theta_l$
  \end{alertblock}
\end{frame}

\begin{frame}{Interacción con el entorno: fuerzas de contacto}
  \begin{block}{Condición de contacto}
    Contacto cuando: $y_l(t) - h_c \leq y_{c0}(x_l, t)$ \quad (perfil de obstáculos)
  \end{block}

  \begin{block}{Fuerza vertical: modelo elástico-amortiguado}
    \[
    F_{cy} = 
    \begin{cases} 
      K_{cy}(y_{c0} - (y_l - h_c)) - b_{cy} v_{ly} & \text{(Apoyado)} \\[4pt]
      0 & \text{(Suspendido)}
    \end{cases}
    \]
    \begin{itemize}
      \item $K_{cy} = 1.8 \times 10^9$ N/m \quad (rigidez)
      \item $b_{cy} = 10 \times 10^6$ Ns/m \quad (amortiguamiento)
    \end{itemize}
  \end{block}

  \begin{block}{Fuerza horizontal: fricción de contacto}
    \[
    F_{cx} = 
    \begin{cases} 
      -b_{cx} v_{lx} & \text{(Apoyado)} \\[4pt]
      0 & \text{(Suspendido)}
    \end{cases}
    \quad ; \quad b_{cx} = 1 \times 10^6 \text{ Ns/m}
    \]
  \end{block}
\end{frame}

\begin{frame}{Acoplamiento elástico del cable}
  \begin{block}{Modelado: resorte-amortiguador con rigidez variable}
    La rigidez y amortiguamiento escalan con la longitud del cable bajo tensión:
    \[
    K_{hw}(l_h) = \frac{k_{hwu}}{2 l_h + L_{h0}} \quad ; \quad b_{hw}(l_h) = b_{hwu}(2 l_h + L_{h0})
    \]
    \begin{itemize}
      \item $k_{hwu} = 236 \times 10^6$ N/m·m (rigidez unitaria)
      \item $b_{hwu} = 150$ Ns/m·m (amortiguamiento unitario)
      \item $L_{h0} = 110$ m (despliegue fijo)
    \end{itemize}
  \end{block}

  \begin{block}{Fuerza de tensión: solo si cable está tenso}
    \[
    F_{hw} = 
    \begin{cases} 
      K_{hw}(l_h) \cdot 2(l - l_h) + b_{hw}(l_h) \cdot 2(\dot{l} - \dot{l}_h) & \text{(Tenso)} \\[4pt]
      0 & \text{(Flojo)}
    \end{cases}
    \]
  \end{block}
\end{frame}

\begin{frame}{Dinámica del accionamiento: estructura}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Eje lento (tambor)}
      \begin{itemize}
        \item Freno de emergencia: $T_{hEb}$
        \item Inercia: $J_{hd+hEb} = 3800$ kgm²
        \item Radio: $r_{hd} = 0.75$ m
        \item Fricción: $b_{hd} = 8$ Nms/rad
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Eje rápido (motor)}
      \begin{itemize}
        \item Freno de operación: $T_{hb}$
        \item Inercia: $J_{hm+hb} = 30$ kgm²
        \item Transmisión: $i_h = 22$
        \item Fricción: $b_{hm} = 18$ Nms/rad
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.3cm}
  \begin{block}{Relación cinemática (aparejo 2:1)}
    \[
    v_h(t) = \frac{r_{hd} \omega_{hd}(t)}{2} = -\frac{dl_h}{dt}
    \]
    Factor 2 divide la velocidad tangencial del tambor
  \end{block}
\end{frame}

\begin{frame}{Frenos mecánicos: operación y emergencia}
  \begin{block}{Lógica NC (Normally Closed) para ambos}
    \begin{itemize}
      \item Energizado ($=1$): freno \alert{abierto}, motor libre
      \item Desenergizado ($=0$): freno \alert{cerrado}, torque máximo
    \end{itemize}
  \end{block}

  \begin{block}{Freno de operación (motor)}
    \[
    T_{hb}(\text{BRK}_h) = 
    \begin{cases} 
      0 & \text{si BRK}_h = 1 \\[4pt]
      -T_{hb\max} \tanh(b_{hb} \omega_{hm}) & \text{si BRK}_h = 0
    \end{cases}
    \]
    \begin{itemize}
      \item $T_{hb\max} = 50 \times 10^3$ Nm \quad (torque máximo)
      \item $b_{hb} = 100 \times 10^6$ Nms/rad \quad (fricción viscosa)
    \end{itemize}
  \end{block}

  \begin{block}{Freno de emergencia (tambor) — Nivel 0}
    \[
    T_{hEb}(\text{BRK}_{hE}) = 
    \begin{cases} 
      0 & \text{si BRK}_{hE} = 1 \\[4pt]
      -T_{hEb\max} \tanh(b_{hEb} \omega_{hd}) & \text{si BRK}_{hE} = 0
    \end{cases}
    \]
    \begin{itemize}
      \item $T_{hEb\max} = 1.1 \times 10^6$ Nm
      \item $b_{hEb} = 2.2 \times 10^9$ Nms/rad
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Modelo equivalente referido al motor}
  \begin{block}{Reducción a un solo eje equivalente}
    Al igual que el carro, se refiere la dinámica del tambor al motor usando $i_h = 22$:
  \end{block}

  \vspace{-0.2cm}
  \[
  J_{h,eq} = J_{hm+hb} + \frac{J_{hd+hEb}}{i_h^2} = 30 + \frac{3800}{22^2} \approx 37.9 \text{ kgm}^2
  \]

  \vspace{-0.3cm}
  \[
  b_{h,eq} = b_{hm} + \frac{b_{hd}}{i_h^2} = 18 + \frac{8}{22^2} \approx 18.016 \text{ Nms/rad}
  \]

  \begin{alertblock}{Resultado}
    Ecuación de estado única para todo el accionamiento:
    \[
    \frac{d\omega_{hm}}{dt} = \frac{1}{J_{h,eq}}\left[T_{hm} + T_{hb} + \frac{T_{hEb}}{i_h} - b_{h,eq}\omega_{hm} - \frac{r_{hd}}{i_h}F_{hw}\right]
    \]
  \end{alertblock}
\end{frame}

\begin{frame}{Modelo dinámico completo: ecuaciones de estado}
  \begin{block}{Sistema de ecuaciones acopladas}
    \begin{enumerate}
      \item \textbf{Motor}: $\frac{d\omega_{hm}}{dt} = f_1(\omega_{hm}, F_{hw}, T_{hm}, \text{BRK}_h, \text{BRK}_{hE})$
      \vspace{0.1cm}
      \item \textbf{Velocidades de carga}: 
      \begin{itemize}
        \item $\frac{dv_{lx}}{dt} = f_2(v_{lx}, \theta_l, F_{hw}, F_{cx}, m_l)$
        \item $\frac{dv_{ly}}{dt} = f_3(v_{ly}, \theta_l, F_{hw}, F_{cy}, m_l)$
      \end{itemize}
      \vspace{0.1cm}
      \item \textbf{Posiciones}: $\frac{dy_l}{dt} = v_{ly}$ ; $\frac{dl_h}{dt} = -\frac{r_{hd}}{2i_h}\omega_{hm}$
      \vspace{0.1cm}
      \item \textbf{Fuerzas}: $F_{hw}(l, l_h, \dot{l}, \dot{l}_h)$ con lógica de cable tenso/flojo
    \end{enumerate}
  \end{block}

  \vspace{0.2cm}
  \begin{alertblock}{Dimensionalidad del sistema de izaje}
    Sistema híbrido con \textbf{4–5 estados continuos} (según el cable) + \textbf{2 eventos discretos} (TLK, contacto)
  \end{alertblock}
\end{frame}
