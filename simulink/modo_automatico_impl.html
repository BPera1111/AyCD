Acciones de estados por fases

1. Selección de Misión
SEL_ORIGEN
entry
i_sel = 1; debounce = 0;
x_inicio = xt_enc;
x_origen = xt_min + (i_lista(1)-1)*Wc + Wc/2;
during
debounce = debounce + 1;
if (debounce > 25)
  if (Joy_CMD_X > dead_zone)
    i_sel = mod(i_sel, length(i_lista)) + 1; debounce=0;
  elseif (Joy_CMD_X < -dead_zone)
    i_sel = mod(i_sel-2, length(i_lista)) + 1; debounce=0;
  end
end
x_origen = xt_min + (i_lista(i_sel)-1)*Wc + Wc/2;
exit
i_origen = i_lista(i_sel);
SEL_DESTINO
entry
i_sel = 1; debounce = 0;
x_destino = xt_min + (i_lista(1)-1)*Wc + Wc/2;
during
% (Misma lógica de debounce y Joy_X)
x_destino = xt_min + (i_lista(i_sel)-1)*Wc + Wc/2;
exit
i_destino = i_lista(i_sel);
2. Ir a buscar el contenedor
SUBIR_INICIAL
entry
y_max_ruta = 0;
for k = min(i_carro,i_origen):max(i_carro,i_origen)
  if yc0_vec(k) > y_max_ruta
    y_max_ruta = yc0_vec(k);
  end
end
y_segura_origen = min(y_max_ruta + Hc + 1.0, yh_max - 1.0);
during
vx_target = 0; vy_target = vh_lim;
trans. salida → IR_ORIGEN
[yh_enc >= y_segura_origen]
IR_ORIGEN
during
err_x = x_origen - xt_enc;
vx_target = max(-vt_max, min(vt_max, 3.0*err_x));
vy_target = 0;
trans. salida → BAJAR_ORIGEN
[abs(xt_enc - x_origen) < 0.15 && abs(vt_obs) < 0.1]
BAJAR_ORIGEN
entry
y_obj = yc0_vec(i_origen) + 0.5;
during
err_y = y_obj - yh_enc;
vy_target = max(-vh_lim, min(vh_lim, 3.0*err_y));
vx_target = 0;
trans. salida → ENGANCHAR
[hay_cont == 1]
ENGANCHAR
entry
vy_target = 0;
cmd_auto_tlk = 1; % Cierra twistlocks
exit
cmd_auto_tlk = 0;
trans. salida → SUBIR_CARGADO
[TLK == 1]
3. Trasladar y dejar
SUBIR_CARGADO
entry
y_max_ruta = 0;
for k = min(i_origen,i_destino):max(i_origen,i_destino)
  if yc0_vec(k) > y_max_ruta
    y_max_ruta = yc0_vec(k);
  end
end
y_segura_destino = min(y_max_ruta + Hc + 1.0, yh_max - 1.0);
during
vx_target = 0; vy_target = vh_lim;
trans. salida → IR_DESTINO
[yh_enc >= y_segura_destino]
IR_DESTINO
during
err_x = x_destino - xt_enc;
vx_target = max(-vt_max, min(vt_max, 3.0*err_x));
vy_target = 0;
trans. salida → BAJAR_DESTINO
[abs(xt_enc - x_destino) < 0.15 && abs(vt_obs) < 0.1]
BAJAR_DESTINO
entry
altura_apoyo = yc0_vec(i_destino) + Hc + 0.5;
during
err_y = altura_apoyo - yh_enc;
vy_target = max(-vh_lim, min(vh_lim, 3.0*err_y));
trans. salida → DESENGANCHAR
[yh_enc <= altura_apoyo || Fhw < (Ms*g/2 * 0.8)]
DESENGANCHAR
entry
vy_target = 0;
cmd_auto_tlk = 1; % Abre twistlocks
exit
cmd_auto_tlk = 0;
trans. salida → SUBIR_VACIO
[TLK == 0]
4. Soltar y volver
SUBIR_VACIO
entry
y_max_ruta = 0;
for k = min(i_destino,i_carro):max(i_destino,i_carro)
  if yc0_vec(k) > y_max_ruta
    y_max_ruta = yc0_vec(k);
  end
end
y_segura_inicio = min(y_max_ruta + Hc + 1.0, yh_max - 1.0);
during
vx_target = 0; vy_target = vh_lim;
trans. salida → VOLVER_INICIO
[yh_enc >= y_segura_inicio]
VOLVER_INICIO
during
err_x = x_inicio - xt_enc;
vx_target = max(-vt_max, min(vt_max, 3.0*err_x));
vy_target = 0;
trans. salida → FIN_MISION
[abs(xt_enc - x_inicio) < 0.15]
FIN_MISION
entry
vx_target = 0;
vy_target = 0;
trans. salida → REPOSO
(Transición directa por defecto)
